// Codice Relay: usa nomi prefissati per non interferire con quelli del mod.
var relayLastMisses:Int = 0;
var relayLastSong:String = null;
var relaySinceLastSend:Float = 0;

function relaySend(missHappened:Bool) {
	if (funkin.game.PlayState.instance == null) return;
	var ps = funkin.game.PlayState.instance;
	var songName = (funkin.game.PlayState.SONG != null && funkin.game.PlayState.SONG.meta != null) ? funkin.game.PlayState.SONG.meta.name : null;
	try {
		var body = haxe.Json.stringify({
			song_name: songName,
			difficulty: funkin.game.PlayState.difficulty,
			score: ps.songScore,
			accuracy: ps.accuracy,
			miss: missHappened,
			sent_at: Date.now().getTime()
		});
		var relayDir = Sys.getEnv("APPDATA") + "/Relay";
		if (!sys.FileSystem.exists(relayDir)) sys.FileSystem.createDirectory(relayDir);
		sys.io.File.saveContent(relayDir + "/fnf-event.json", body);
		var http = new haxe.Http("http://127.0.0.1:__PORT__/event");
		http.setHeader("Content-Type", "application/json");
		http.setPostData(body);
		http.onError = function(e:String) {};
		http.request(true);
	} catch (e:Dynamic) {}
}

function __CALLBACK__(elapsed:Float) {
	if (funkin.game.PlayState.instance == null) return;
	var ps = funkin.game.PlayState.instance;
	var songName = (funkin.game.PlayState.SONG != null && funkin.game.PlayState.SONG.meta != null) ? funkin.game.PlayState.SONG.meta.name : null;
	if (songName != relayLastSong) {
		relayLastSong = songName;
		relayLastMisses = ps.misses;
		relaySinceLastSend = 2;
	}
	if (ps.misses != relayLastMisses) {
		relayLastMisses = ps.misses;
		relaySend(true);
		relaySinceLastSend = 0;
		return;
	}
	relaySinceLastSend += elapsed;
	if (relaySinceLastSend >= 2.0) {
		relaySinceLastSend = 0;
		relaySend(false);
	}
}
