-- Relay: registra in automatico i tuoi record. Scritto e aggiornato dall'app Relay, non modificarlo.
-- Gli eventi vanno in relay/events.jsonl nella cartella del gioco, che l'app legge.
-- Funziona con Psych Engine 0.5, 0.6, 0.7 e 1.0. Nei giochi che non caricano gli script di
-- mods/scripts (per esempio Mario's Madness) sta in fondo allo script di ogni canzone: per questo
-- richiama sempre le funzioni che lo script della canzone aveva gia' definito.

local relayPrev = {
  onCreatePost = onCreatePost,
  onSongStart = onSongStart,
  onEndSong = onEndSong,
  onGameOver = onGameOver,
  onDestroy = onDestroy,
}
local relayPlaying = false
local relayEnded = false

local function relayQuote(s)
  s = tostring(s or '')
  s = s:gsub('\\', '\\\\'):gsub('"', '\\"'):gsub('[%c]', ' ')
  return '"' .. s .. '"'
end

local function relayProp(name)
  local ok, v = pcall(getProperty, name)
  if ok then return v end
  return nil
end

local function relayClassProp(name)
  if getPropertyFromClass == nil then return nil end
  local ok, v = pcall(getPropertyFromClass, 'PlayState', name)
  if ok then return v end
  return nil
end

-- Niente record in botplay o in practice mode (nelle versioni vecchie di Psych sono variabili
-- statiche di PlayState). La modalita' chart editor non si controlla: alcune mod (Pibby
-- Apocalypse) la lasciano sempre accesa per errore.
local function relayValid()
  local bot = relayProp('cpuControlled') == true or relayClassProp('cpuControlled') == true or botPlay == true
  local practice = relayProp('practiceMode') == true or relayClassProp('practiceMode') == true
  return not (bot or practice)
end

local function relayDifficulty()
  if difficultyName ~= nil and difficultyName ~= '' then return difficultyName end
  local text = relayProp('storyDifficultyText')
  if text ~= nil and text ~= '' then return text end
  return ''
end

local function relayEmit(kind, extra)
  local parts = {
    '"kind":' .. relayQuote(kind),
    '"engine":"psych"',
    '"song_name":' .. relayQuote(songName),
    '"difficulty":' .. relayQuote(relayDifficulty()),
    '"mod_folder":' .. relayQuote(currentModDirectory),
    '"valid":' .. tostring(relayValid()),
  }
  for k, v in pairs(extra or {}) do
    parts[#parts + 1] = '"' .. k .. '":' .. v
  end
  local ok, err = pcall(function()
    local f = io.open('relay/events.jsonl', 'a')
    if f then
      f:write('{' .. table.concat(parts, ',') .. '}\n')
      f:close()
    end
  end)
  if not ok then debugPrint('[Relay] evento non scritto: ' .. tostring(err)) end
end

local function relayThen(name, ...)
  local prev = relayPrev[name]
  if prev ~= nil then return prev(...) end
  return Function_Continue
end

function onCreatePost(...)
  relayPlaying = false
  relayEnded = false
  relayEmit('song_load')
  return relayThen('onCreatePost', ...)
end

function onSongStart(...)
  relayPlaying = true
  relayEmit('song_start')
  return relayThen('onSongStart', ...)
end

function onEndSong(...)
  if relayPlaying and not relayEnded then
    relayEnded = true
    relayPlaying = false
    local score = tonumber(relayProp('songScore')) or 0
    local misses = tonumber(relayProp('songMisses')) or 0
    local rating = tonumber(relayProp('ratingPercent'))
    local extra = { score = tostring(math.floor(score)), misses = tostring(math.floor(misses)) }
    if rating ~= nil and rating == rating then extra.accuracy = tostring(rating) end
    relayEmit('song_end', extra)
  end
  return relayThen('onEndSong', ...)
end

function onGameOver(...)
  if relayPlaying then
    relayPlaying = false
    relayEmit('song_abort', { reason = relayQuote('game over') })
  end
  return relayThen('onGameOver', ...)
end

function onDestroy(...)
  if relayPlaying then
    relayPlaying = false
    relayEmit('song_abort', { reason = relayQuote('uscita o canzone ricominciata') })
  end
  return relayThen('onDestroy', ...)
end
